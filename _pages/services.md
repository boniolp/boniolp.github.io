/* Turns each "[Venue (ACRONYM) year](link): track **role**" item into a row (acronym tag, venue, track, role),
   shows journal / reviewer lists as a grid, and adds counts. Without JavaScript the plain (styled) lists are shown. */
(function () {
  var root = document.querySelector('.sv');
  if (!root) return;
  function esc(s) { return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }
  var section = null, stats = [], prev = null;
  Array.prototype.forEach.call(root.children, function (el) {
    if (el.tagName === 'H2') { section = { h: el, n: 0, years: {} }; stats.push(section); }
    if (el.tagName === 'UL' && section) {
      var grid = prev && prev.tagName === 'H2';
      el.className = grid ? 'sv-grid' : 'sv-rows';
      Array.prototype.forEach.call(el.children, function (li) {
        section.n++;
        if (prev && prev.tagName === 'H4') prev._n = (prev._n || 0) + 1;
        if (grid) return;
        var box = li.querySelector('p') || li, venue = null, track = '', role = '', btns = [];
        Array.prototype.slice.call(box.childNodes).forEach(function (n) {
          if (n.tagName === 'A' && n.querySelector('img')) {
            btns.push('<a class="sv-btn" href="' + n.getAttribute('href') + '">' + esc(n.querySelector('img').getAttribute('alt') || 'Link') + '</a>');
          } else if (n.tagName === 'A' && !venue) { venue = n; }
          else if (n.tagName === 'STRONG') { role = n.textContent; }
          else if (venue) { track += n.textContent; }
        });
        if (!venue) return;
        var name = venue.textContent.trim(), m = name.match(/\(([^()]+)\)/);
        var acr = m ? m[1].replace(/\s*\d{4}$/, '') : '';
        track = track.replace(/^[\s:]+/, '').replace(/\s+$/, '');
        var vh = venue.getAttribute('href') ? '<a href="' + venue.getAttribute('href') + '">' + esc(name) + '</a>' : esc(name);
        li.innerHTML = (acr ? '<span class="sv-acr">' + esc(acr) + '</span>' : '<span class="sv-acr sv-none" style="visibility:hidden"></span>') +
                       '<span class="sv-venue">' + vh + '</span>' +
                       (track ? '<span class="sv-track">' + esc(track) + '</span>' : '') +
                       (role ? '<span class="sv-role">' + esc(role) + '</span>' : '') + btns.join('');
      });
    }
    if (el.tagName === 'H2' || el.tagName === 'H4' || el.tagName === 'UL') prev = el;
  });
  root.querySelectorAll('h4').forEach(function (h) {
    if (h._n) { var c = document.createElement('span'); c.className = 'sv-count'; c.textContent = h._n; h.appendChild(c); }
  });
  var bar = document.createElement('div'); bar.className = 'sv-summary';
  stats.forEach(function (s) {
    var c = document.createElement('span'); c.className = 'sv-count'; c.textContent = s.n; s.h.appendChild(c);
    bar.insertAdjacentHTML('beforeend', '<span class="sv-stat"><b>' + s.n + '</b>' + esc(s.h.firstChild.textContent.trim()) + '</span>');
  });
  var h1 = root.querySelector('h1');
  root.insertBefore(bar, h1 ? h1.nextSibling : root.firstChild);
})();
</script>
